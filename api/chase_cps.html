CHASE: Contract-based Heterogeneous Analysis and Systems Exploration.

Python interface to the CHASE core library and its back ends::

from chase_cps import Contract, refinement

spec = Contract.from_strings(inputs=["req"], outputs=["grant"],
                             assumptions=["G F !req"],
                             guarantees=["G (req -> F grant)"], name="Spec")
impl = Contract.from_strings(inputs=["req"], outputs=["grant"],
                             guarantees=["G (req -> X grant)"], name="Impl")
result = refinement(impl, spec)    # valid, or invalid with a counterexample

 57class Contract:
 58    """An assume-guarantee contract.
 59
 60    ``assumptions`` and ``guarantees`` are lists of LTL formulas, read as
 61    conjunctions. Variables are Boolean ("bool") or bounded integers
 62    (low, high).
 63    """
 64
 65    name: str
 66    inputs: dict[str, VarType] = field(default_factory=dict)
 67    outputs: dict[str, VarType] = field(default_factory=dict)
 68    assumptions: list[ltl.Node] = field(default_factory=list)
 69    guarantees: list[ltl.Node] = field(default_factory=list)

 71    # Construction ------------------------------------------------------------

 73    @classmethod
 74    def from_strings(cls, inputs=None, outputs=None, assumptions=None,
 75                     guarantees=None, name: str = "C") -> "Contract":
 76        """Build a contract from formulas in the standard LTL syntax.
 77
 78        >>> c = Contract.from_strings(inputs=["req"], outputs=["grant"],
 79        ...                           assumptions=["G F !req"],
 80        ...                           guarantees=["G (req -> F grant)"])
 81        """
 82        c = cls(name=name, inputs=_normalize_vars(inputs),
 83                outputs=_normalize_vars(outputs),
 84                assumptions=_normalize_formulas(assumptions),
 85                guarantees=_normalize_formulas(guarantees))
 86        c.validate()
 87        return c

 89    @classmethod
 90    def from_pychase(cls, pc) -> "Contract":
 91        """Convert a contract of the CHASE core library."""
 92        from ._native import representation as R
 93        inputs, outputs, generic, constants = {}, {}, {}, {}
 94        for d in pc.declarations:
 95            name = d.getName().getString()
 96            if type(d).__name__ == "Constant":
 97                constants[name] = d.getValue().getValue()
 98                continue
 99            t = d.getType()
100            vtype = (t.getMin(), t.getMax()) if type(t).__name__ == "Integer" else "bool"
101            causality = d.getCausality()
102            if causality == R.causality_t.input:
103                inputs[name] = vtype
104            elif causality == R.causality_t.generic:   # e.g. a proposition, inlined
105                generic[name] = vtype
106            else:
107                outputs[name] = vtype
108
109        def value(node):    # constants are replaced by their values
110            if isinstance(node, ltl.Var) and node.name in constants:
111                return ltl.Num(constants[node.name])
112            if isinstance(node, ltl.Unary):
113                return ltl.Unary(node.op, value(node.arg))
114            if isinstance(node, ltl.Binary):
115                return ltl.Binary(node.op, value(node.left), value(node.right))
116            return node
117
118        def terms(domain_map) -> list[ltl.Node]:
119            out = []
120            for spec in domain_map.values():
121                out.extend(value(t) for t in ltl.conjuncts(ltl.from_pychase(spec)))
122            return out
123
124        assumptions, guarantees = terms(pc.assumptions), terms(pc.guarantees)
125        used = ltl.variables(ltl.conjunction(assumptions + guarantees))
126        outputs.update({k: v for k, v in generic.items() if k in used})
127        return cls(name=pc.getName().getString(), inputs=inputs, outputs=outputs,
128                   assumptions=assumptions, guarantees=guarantees)
129
130    def validate(self) -> None:
131        for n in [self.name] + list(self.variables):
132            if n in RESERVED or not n.replace("_", "a").isalnum() or not n[0].isalpha():
133                raise ValueError(f"{n!r} cannot be used as a name: names start with a "
134                                 "letter, contain letters, digits and _, and are not "
135                                 f"keywords of CHASE ({', '.join(sorted(RESERVED))})")
136        both = set(self.inputs) & set(self.outputs)
137        if both:
138            raise ValueError(f"variables both input and output: {sorted(both)}")
139        declared = self.variables
140        for f in self.assumptions + self.guarantees:
141            missing = ltl.variables(f) - set(declared)
142            if missing:
143                raise ValueError(f"undeclared variables {sorted(missing)} in "
144                                 f"{ltl.to_str(f)!r} (contract {self.name})")

186    def __str__(self) -> str:
187        lines = [f"CONTRACT {self.name}"]
188        for k, v in self.variables.items():
189            kind = "input" if k in self.inputs else "output"
190            lines.append(f"  {kind} {k} : {'bool' if v == 'bool' else f'{v[0]}..{v[1]}'}")
191        lines.append("  Assumptions:")
192        lines += [f"    {ltl.to_str(f)}" for f in self.assumptions] or ["    true"]
193        lines.append("  Guarantees:")
194        lines += [f"    {ltl.to_str(f)}" for f in self.guarantees] or ["    true"]
195        return "\n".join(lines)

199    def to_logics(self) -> str:
200        """The contract in the logics language of CHASE (as read by logics_tool)."""
201        lines = [f"CONTRACT {self.name}:"]
202        for k, v in self.variables.items():
203            kind = "input" if k in self.inputs else "output"
204            typ = "boolean" if v == "bool" else f"integer ({v[0]}:{v[1]})"
205            lines.append(f"    {kind} {typ} variable {k};")
206        lines.append("    Assumptions:")
207        lines += [f"        {ltl.to_chase(f)};" for f in self.assumptions] or ["        true;"]
208        lines.append("    Guarantees:")
209        lines += [f"        {ltl.to_chase(f)};" for f in self.guarantees] or ["        true;"]
210        return "\n".join(lines) + "\n"

212    def to_pychase(self):
213        """Parse the contract with the CHASE core library and return the
214        pychase ``Contract`` (the CHASE objects are never freed)."""
215        return _parse_contracts([self])[self.name]

219    def rename(self, mapping: Mapping[str, str], name: str | None = None) -> "Contract":
220        """The same contract with the variables renamed (old name -> new name)."""
221        def ren(node):
222            if isinstance(node, ltl.Var):
223                return ltl.Var(mapping.get(node.name, node.name), node.primed)
224            if isinstance(node, ltl.Unary):
225                return ltl.Unary(node.op, ren(node.arg))
226            if isinstance(node, ltl.Binary):
227                return ltl.Binary(node.op, ren(node.left), ren(node.right))
228            return node
229        c = Contract(name or self.name,
230                     {mapping.get(k, k): v for k, v in self.inputs.items()},
231                     {mapping.get(k, k): v for k, v in self.outputs.items()},
232                     [ren(f) for f in self.assumptions], [ren(f) for f in self.guarantees])
233        c.validate()
234        return c

236    def shares_io_with(self, other: "Contract") -> bool:
237        """Same inputs and outputs, with the same types."""
238        return self.inputs == other.inputs and self.outputs == other.outputs
239
240    def can_compose_with(self, other: "Contract") -> bool:
241        """No variable is an output of both contracts, and the shared
242        variables have the same type."""
243        if set(self.outputs) & set(other.outputs):
244            return False
245        return all(self.variables[v] == other.variables[v]
246                   for v in set(self.variables) & set(other.variables))

248    # Contract algebra (computed by the CHASE core library) -------------------
249
250    def simplify(self, name: str | None = None, nots: bool = True,
251                 temporal_operators: bool = True) -> "Contract":
252        """The contract with its formulas simplified by the core library (the
253        simplify command of logics_tool): the negations are moved towards the
254        atoms (nots), and the temporal operators are grouped
255        (temporal_operators): G f & G g becomes G (f & g), F f | F g becomes
256        F (f | g), X f & X g becomes X (f & g). The contract is equivalent."""
257        from ._native import utilities
258        pc = self.to_pychase()
259        utilities.simplify(pc, utilities.simplify_options(nots, temporal_operators))
260        out = Contract.from_pychase(pc)
261        out.name = name or self.name
262        out.inputs = {v: t for v, t in out.variables.items() if v in self.inputs}
263        out.outputs = {v: t for v, t in out.variables.items() if v not in self.inputs}
264        return out

266    def saturate(self, name: str | None = None) -> "Contract":
267        """Saturation: the equivalent contract (A, A -> G)."""
268        guarantees = [self.saturated_guarantee] if self.assumptions else list(self.guarantees)
269        return Contract(name or self.name, dict(self.inputs), dict(self.outputs),
270                        list(self.assumptions), guarantees)
271
272    def compose(self, other: "Contract", name: str | None = None) -> "Contract":
273        """Parallel composition; variables with the same name are connected."""
274        both = set(self.outputs) & set(other.outputs)
275        if both:
276            raise ValueError(f"cannot compose {self.name} and {other.name}: "
277                             f"{sorted(both)} are outputs of both")
278        return _algebra("composition", self, other, name or f"{self.name}_{other.name}")
279
280    def conjunction(self, other: "Contract", name: str | None = None) -> "Contract":
281        """Greatest lower bound (both specifications at once):
282        (A1 | A2, G1 & G2)."""
283        return _algebra("conjunction", self, other, name or f"{self.name}_and_{other.name}")
284
285    def disjunction(self, other: "Contract", name: str | None = None) -> "Contract":
286        """Least upper bound: (A1 & A2, G1 | G2)."""
287        return _algebra("disjunction", self, other, name or f"{self.name}_or_{other.name}")
288
289    def merging(self, other: "Contract", name: str | None = None) -> "Contract":
290        """Merging of two viewpoints of the same component:
291        (A1 & A2, (G1 & G2) | !(A1 & A2))."""
292        return _algebra("merging", self, other, name or f"{self.name}_merge_{other.name}")
293
294    def quotient(self, other: "Contract", name: str | None = None) -> "Contract":
295        """Largest C such that other || C refines self:
296        (A1 & G2, (G1 & A2) | !(A1 & G2)) (Incer et al., MEMOCODE 2018)."""
297        return _algebra("quotient", self, other, name or f"{self.name}_div_{other.name}")
298
299    def separation(self, other: "Contract", name: str | None = None) -> "Contract":
300        """Smallest C such that self refines other . C (merging):
301        ((A1 & G2) | !(G1 & A2), G1 & A2) (Passerone et al., TECS 2019)."""
302        return _algebra("separation", self, other, name or f"{self.name}_sep_{other.name}")

def remove_redundant( contract: Contract, backend: str = 'auto', **options) -> Redundancy: 
152def remove_redundant(contract: Contract, backend: str = "auto", **options) -> Redundancy:
153    """Drop each term implied by the other terms and the context.
154
155    An assumption is dropped when the other assumptions imply it; a guarantee
156    when the assumptions and the other guarantees imply it. Each test is one
157    implication check; a term whose check does not finish is kept.
158    """
159    c = contract
160    variables = c.variables
161    out = Redundancy(Contract(c.name, dict(c.inputs), dict(c.outputs),
162                                  list(c.assumptions), list(c.guarantees)))
163    kept = list(c.assumptions)
164    for term in list(kept):
165        others = [t for t in kept if t is not term]
166        r = implies(ltl.conjunction(others), term, variables, backend, **options)
167        if r.status == VALID:
168            kept.remove(term)
169            out.removed_assumptions.append(term)
170        elif r.status == UNKNOWN:
171            out.unknown.append(term)
172    out.contract.assumptions = kept
173    context = ltl.conjunction(kept)
174    kept_g = list(c.guarantees)
175    for term in list(kept_g):
176        others = [t for t in kept_g if t is not term]
177        r = implies(ltl.conjunction([context] + others), term, variables, backend, **options)
178        if r.status == VALID:
179            kept_g.remove(term)
180            out.removed_guarantees.append(term)
181        elif r.status == UNKNOWN:
182            out.unknown.append(term)
183    out.contract.guarantees = kept_g
184    return out
