chase_cps.viz

Diagrams of contracts, components, and systems.

Four rules:

  • A contract is a rounded box. Its assumptions and guarantees open under its name.
  • A component with a model (its implementation: RTL, a state machine in JSON or SMV, a synthesized controller) is a square box: its contract, then its model, whose states and transitions open under it, and whether the model implements the contract.
  • Refinement is drawn as nesting: what refines a contract is drawn inside its box, a decomposition (contracts or components side by side, connected by the names of their variables) or a single contract, with whether it refines the contract.
  • A quotient is a dashed hole next to the known parts; a contract that refines the quotient is drawn inside the hole. Merging and conjunction stack their operands in one box.

Around them: an environment is a box on the left that drives the inputs of what it is the environment of; the results of the checks of a box are marks under its name (compatible, consistent, realizable); a core is marked on the terms of the contract; the context of the checks is written above the diagram.

The structure comes from the contracts themselves: each result of the algebra remembers how it was computed (Contract.origin). The page is drawn in the browser, laid out by ELK (the Eclipse Layout Kernel, loaded from a CDN).

viz.save(viz.diagram(system, refines=[(text, q3, "valid")]), "plan.html")

            environments=[(env, c, is_environment(env, c))], context="G !(a & b)")

 114    def contract(self, c: Contract, title: Optional[str] = None) -> dict:
 115        origin = c.origin
 116        if not origin or origin[0] not in SIGNS:
 117            note = None
 118            if origin:
 119                note = f"{origin[0]} of " + ", ".join(o.name for o in origin[1])
 120            return self.box(c, types=_types(c), kind="contract", title=title or c.name, note=note,
 121                            contract=self.terms(c), inputs=sorted(c.inputs),
 122                            outputs=sorted(c.outputs))
 123        operation, operands = origin
 124        if operation == "composition":
 125            parts = []
 126            for o in operands:            # a chain of compositions is one level
 127                node = self.contract(o)
 128                parts += node["children"] if node.get("op") == "composition" else [node]
 129            return self.group(c, operation, parts)
 130        if operation == "quotient":
 131            dividend, divisor = operands
 132            known = self.contract(divisor)
 133            known = known["children"] if known.get("op") == "composition" else [known]
 134            hole = self.box(c, types=_types(c), kind="hole", title=c.name,
 135                            note=f"{dividend.name} / {divisor.name}",
 136                            contract=self.terms(c), inputs=sorted(c.inputs),
 137                            outputs=sorted(c.outputs))
 138            node = self.group(dividend, operation, known + [hole], box_of=c)
 139            node["note"] = " ⊗ ".join([k["title"] for k in known] + ["?"])
 140            node["contract"] = self.terms(dividend)
 141            return node
 142        return self.group(c, operation, [self.contract(o) for o in operands])

 144    def group(self, c: Contract, operation: str, children: list, box_of=None) -> dict:
 145        names = [ch["title"] for ch in children]
 146        if operation in STACKED:      # viewpoints of one component: its ports, no wires inside
 147            for ch in children:         # a viewpoint shares the ports of the box
 148                if not ch.get("edges"):
 149                    ch["bare"] = True
 150            parts = [ch for ch in children if not ch.get("bare")]   # decompositions: wired to the box
 151            outputs = sorted(c.outputs)
 152            return self.box(box_of if box_of is not None else c, types=_types(c), kind="group",
 153                            op=operation,
 154                            title=c.name, note=f" {SIGNS[operation]} ".join(names),
 155                            layout="stack", contract=self.terms(c), inputs=sorted(c.inputs),
 156                            outputs=outputs, children=children,
 157                            edges=[w for ch in parts for w in _through(ch, c.inputs, outputs)])
 158        outputs = _outside(children, c.outputs)
 159        return self.box(box_of if box_of is not None else c, types=_types(c), kind="group",
 160                        op=operation, title=c.name, note=f" {SIGNS[operation]} ".join(names),
 161                        layout="row", contract=self.terms(c), inputs=sorted(c.inputs), outputs=outputs,
 162                        children=children, edges=_wires(children, c.inputs) + _out(children, outputs))

 166    def component(self, k: Component) -> dict:
 167        if k.implementation is None:          # a contract with a name
 168            node = self.contract(k.contract, title=k.name)
 169            self.ids.setdefault(id(k), node["id"])
 170            return node
 171        fsm = None
 172        try:
 173            from .explicit import state_graph
 174            fsm = state_graph(k.implementation, inputs=k.contract.inputs)
 175        except Exception:                     # the diagram does without it
 176            fsm = None
 177        name = getattr(k.implementation, "name", None) or type(k.implementation).__name__
 178        source = (getattr(k.implementation, "source", None) or {}).get("kind")
 179        node = self.box(k, types=_types(k.contract), kind="component", title=k.name,
 180                        contract=self.terms(k.contract),
 181                        model={"name": name, "fsm": fsm, "source": source},
 182                        badge={"text": "M ⊨ C", "status": k.status},
 183                        inputs=sorted(k.contract.inputs), outputs=sorted(k.contract.outputs))
 184        self.ids.setdefault(id(k.contract), self.ids[id(k)])
 185        return node

 187    def system(self, s: System) -> dict:
 188        children = [self.component(k) for k in s.components]
 189        if s.contract is not None:
 190            inputs = s.contract.inputs
 191        else:
 192            driven = {v for k in s.components for v in k.contract.outputs}
 193            inputs = {v for k in s.components for v in k.contract.inputs if v not in driven}
 194        outputs = _outside(children, s.contract.outputs if s.contract else None)
 195        return self.box(s, types=_types(*(k.contract for k in s.components),
 196                                         *([s.contract] if s.contract else [])),
 197                        kind="group", op="system", title=s.name,
 198                        note=" ⊗ ".join(k.name for k in s.components), contract=None,
 199                        inputs=sorted(inputs), outputs=outputs, children=children,
 200                        edges=_wires(children, inputs) + _out(children, outputs))

 202    def environment(self, env, target: dict, status) -> dict:
 203        """An environment of the box target: it drives the inputs of target."""
 204        from .models import as_model
 205        name, impl = "Environment", env
 206        if isinstance(env, Component):
 207            name, impl = env.name, env.implementation

 209        node = dict(kind="environment", title=name, inputs=[], outputs=list(target.get("inputs", [])),

 211        if model is None:                   # a formula
 212            from .logic import _formula

 214        else:
 215            fsm = None

 217                from .explicit import state_graph
 218                fsm = state_graph(model, inputs=target.get("outputs", []))
 219            except Exception:
 220                fsm = None
 221            node["model"] = {"name": getattr(model, "name", None) or "model", "fsm": fsm,
 222                             "source": (getattr(model, "source", None) or {}).get("kind")}
 223        return self.box(env, types=target.get("types"), **node)

 236    def wrap(self, inner: dict, high: Contract, status) -> dict:
 237        """inner, drawn inside the box of the contract it refines."""
 238        if inner["kind"] == "group":          # a decomposition: the group is the box
 239            if inner.get("contract"):         # the computed composition stays, under its own icon
 240                inner["composed"] = inner["contract"]
 241            outputs = _outside(inner["children"], high.outputs)   # the ports of the contract it refines
 242            inner.update(kind="group", title=high.name, contract=self.terms(high),
 243                         badge={"text": "⪯", "status": status}, outputs=outputs,
 244                         edges=[w for w in inner["edges"] if w.get("to", [0])[0] is not None]
 245                         + _out(inner["children"], outputs))
 246            self.ids.setdefault(id(high), inner["id"])
 247            return inner
 248        return self.box(high, types=_types(high), kind="contract", title=high.name,
 249                        contract=self.terms(high), inputs=sorted(high.inputs), outputs=sorted(high.outputs),
 250                        children=[inner], edges=_through(inner, high.inputs, high.outputs),
 251                        badge={"text": "⪯", "status": status})

 254def _wires(children: list, inputs) -> list:
 255    """The connections inside a box: an output of a child to the inputs of
 256    the others with the same name; an input of the box to the children that
 257    read it. An input read by a child that nobody drives is marked open."""
 258    wires = []
 259    drivers = {v: ch["id"] for ch in children for v in ch.get("outputs", [])}
 260    for ch in children:
 261        for v in ch.get("inputs", []):
 262            if v in drivers and drivers[v] != ch["id"]:
 263                wires.append({"from": [drivers[v], "out", v], "to": [ch["id"], "in", v], "var": v})
 264            elif v in inputs:
 265                wires.append({"from": [None, "in", v], "to": [ch["id"], "in", v], "var": v})
 266            else:
 267                wires.append({"open": [ch["id"], "in", v], "var": v})
 268    return wires

 271def _outside(children: list, outputs=None) -> list:
 272    """The outputs of the box that holds the children: those of its contract
 273    (all of them when there is none), as the children drive them; a signal
 274    between two children is an output too, as in the composed contract."""
 275    driven = {v for ch in children for v in ch.get("outputs", [])}
 276    return sorted(driven if outputs is None else driven & set(outputs))

 279def _out(children: list, outputs) -> list:
 280    """The wires from the outputs of the children to those of the box."""
 281    return [{"from": [ch["id"], "out", v], "to": [None, "out", v], "var": v}
 282            for ch in children for v in ch.get("outputs", []) if v in outputs]

 285def _through(inner: dict, inputs, outputs) -> list:
 286    """The wires from the ports of a box to the box inside it, by name."""
 287    wires = [{"from": [None, "in", v], "to": [inner["id"], "in", v], "var": v}
 288             for v in inner.get("inputs", []) if v in inputs]
 289    wires += [{"from": [inner["id"], "out", v], "to": [None, "out", v], "var": v}
 290              for v in inner.get("outputs", []) if v in outputs]
 291    return wires

 294def _ancestors(nodes: list, node_id: str, path: tuple = ()) -> Optional[list]:
 295    """The boxes that hold the box node_id, the outer first ([] at the top),
 296    or None when it is not drawn."""

 298        if n["id"] == node_id:

 300        found = _ancestors(n.get("children", []), node_id, path + (n,))

 335def _mark(result) -> str:
 336    """valid, invalid, or unknown: whether the check holds."""
 337    status = _status(result)
 338    if status in (None, "unknown"):
 339        return "unknown"
 340    if status in ("vacuous", "not_vacuous"):          # the check holds when nothing is vacuous
 341        return "valid" if status == "not_vacuous" else "invalid"
 342    if isinstance(result, str):
 343        return "valid" if status in ("valid", "sat", "realizable", "separated", "receptive",
 344                                     "found") else "invalid"
 345    return "valid" if bool(result) else "invalid"

 348CHECKS = ("verify, is_realizable, is_well_separated, is_vacuous, or is_receptive; is_compatible "
 349          "and is_consistent with their name, (item, 'consistent', result)")
 350# The label of each check, by the name of the check (for a result that is unknown)
 351KINDS = {"realizable": "R", "well_separated": "WS", "vacuous": "¬V", "receptive": "receptive",
 352         "compatible": "A", "consistent": "G"}

 355def _checks(r, kind: Optional[str] = None) -> list:
 356    """[(label, result, core)]: verify gives A (compatible) and G
 357    (consistent), and A∧G (feasible) when the two cannot hold together;
 358    is_realizable R, is_well_separated WS, is_vacuous ¬V (no vacuous output),
 359    is_receptive receptive. The core explains a failure: a Core, the terms
 360    (A, G) of a vacuous output (its guarantees, and the assumptions of its
 361    core), or None."""
 362    from .checks import Verification
 363    from .gr1 import OutputVacuity
 364    from .logic import Core
 365
 366    def core_of(x):
 367        c = getattr(x, "core", None)
 368        return c if isinstance(c, Core) and c else None
 369    if isinstance(r, Verification):
 370        out = [("A", r.compatible, core_of(r.compatible)), ("G", r.consistent, core_of(r.consistent))]
 371        if r.feasible is not None and not r.feasible:   # when they hold, A & G implies both
 372            out.insert(0, ("A∧G", r.feasible, core_of(r.feasible)))
 373        return out
 374    if isinstance(r, OutputVacuity):
 375        terms = None
 376        if r.vacuous:
 377            a = [t for v in r.vacuous if v.core and v.core.core for t in v.core.core]
 378            terms = ([ltl.to_str(t) for t in a], [ltl.to_str(t) for v in r.vacuous for t in v.guarantees])
 379        return [("¬V", r, terms)]
 380    if kind is not None and kind not in KINDS:
 381        raise ValueError(f"no check {kind!r}: the checks are {', '.join(KINDS)} (and verify)")
 382    label = KINDS.get(kind) or {"realizable": "R", "unrealizable": "R", "separated": "WS",
 383                                "not_separated": "WS", "receptive": "receptive",
 384                                "not_receptive": "receptive"}.get(_status(r))
 385    if label is None or not isinstance(r, Result):
 386        raise TypeError(f"checks takes the results of {CHECKS}, not {r!r}"
 387                        + ("; an unknown result needs the name of its check, as in "
 388                           "(item, 'realizable', result)" if _status(r) == "unknown" else ""))
 389    return [(label, r, core_of(r))]

 392def _core(node: dict, core, unrealizable: bool) -> Optional[dict]:
 393    """Where the terms of the core are in the box: in its contract or in its
 394    composed contract, by index. All the assumptions are in an unrealizable
 395    core: only its guarantees count. The core is a Core, or the texts of its
 396    assumptions and guarantees."""
 397    if isinstance(core, tuple):
 398        want = {"A": set(core[0]), "G": set(core[1])}
 399    elif isinstance(core.core, Contract):
 400        c = core.core
 401        want = {"A": set() if unrealizable else {ltl.to_str(t) for t in c.assumptions},
 402                "G": {ltl.to_str(t) for t in c.guarantees}}
 403    else:                                   # formulas, assumptions or guarantees
 404        fs = {ltl.to_str(t) for t in core.core or []}
 405        want = {"A": fs, "G": fs}
 406    for key in ("contract", "composed"):
 407        terms = node.get(key)
 408        if not terms:
 409            continue
 410        hit = {p: [i for i, t in enumerate(terms[p]) if t in want[p]] for p in "AG"}
 411        found = {terms[p][i] for p in "AG" for i in hit[p]}
 412        if found and found >= want["A"] | want["G"]:
 413            return dict(hit, part=key)
 414    return None

 417def diagram(*items, refines: Iterable = (), checks: Iterable = (), environments: Iterable = (),
 418            context=None, open: Iterable = (), title: Optional[str] = None) -> dict:
 419    """The diagram of contracts, components, and systems, as data for the
 420    page. ``refines`` lists (refined, abstract) or (refined, abstract,
 421    status), where the status is a string or the result of ``refines``: the
 422    refined one is drawn inside the abstract one, or inside the hole when
 423    the abstract one is a quotient. A system with a contract is drawn inside
 424    its contract.
 425
 426    ``checks`` lists (item, result): the result of ``verify`` (A, G, and
 427    A∧G when they cannot hold together), ``is_realizable`` (R),
 428    ``is_well_separated`` (WS), ``is_vacuous`` (¬V), or ``is_receptive``;
 429    with core=True, the core comes with the check that fails. A result that
 430    is unknown does not say which check it is: (item, "realizable", result),
 431    or well_separated, vacuous, receptive; the results of ``is_compatible``
 432    and ``is_consistent`` (sat or unsat) always come with their name,
 433    (item, "consistent", result). ``environments`` lists (environment, item) or
 434    (environment, item, status): an LTL formula, a Model, or a Component
 435    whose implementation is the environment, drawn on the left of the item
 436    and driving its inputs; the status is that of ``is_environment``. The
 437    ``context`` of the checks (a formula or a list of formulas) is written
 438    above the diagram. The items in ``open`` are drawn open: their contract,
 439    their model when it is small, and the core of a check that fails."""
 440    scene = _Scene()
 441    nodes, top, own = [], {}, {}           # own: the box an item drew, before any nesting
 442
 443    def add(obj):
 444        if id(obj) not in top:
 445            node = scene.node(obj)
 446            top[id(obj)] = own[id(obj)] = node
 447            nodes.append(node)
 448        return top[id(obj)]
 449
 450    def replace(old, new):
 451        for i, n in enumerate(nodes):
 452            if n is old:
 453                nodes[i] = new
 454                return
 455        nodes.append(new)
 456
 457    for item in items:
 458        node = add(item)
 459        if isinstance(item, System) and item.contract is not None:
 460            wrapped = scene.wrap(node, item.contract, item.status)
 461            wrapped.update(title=item.name, note=f"contract {item.contract.name} · {node['note']}")
 462            replace(node, wrapped)
 463    def after(box, new):                # new, next to the item that holds box
 464        i = next((i for i, n in enumerate(nodes) if n is box or _find([n], box["id"]) is not None), None)
 465        nodes.insert(len(nodes) if i is None else i + 1, new)
 466
 467    def parts(obj, out):                 # the contracts drawn in the box of obj
 468        out.add(id(obj))
 469        if isinstance(obj, System):
 470            for k in obj.components:
 471                parts(k.contract, out)
 472        elif isinstance(obj, Component):
 473            parts(obj.contract, out)
 474        elif isinstance(obj, Contract) and obj.origin:
 475            for o in obj.origin[1]:
 476                parts(o, out)
 477        return out
 478    # from the top down: a contract is refined after the refinement that draws it
 479    todo, ordered = list(refines), []
 480    while todo:
 481        first = next((r for r in todo
 482                      if not any(x is not r and id(r[1]) in parts(x[0], set()) for x in todo)), todo[0])
 483        ordered.append(first)
 484        todo.remove(first)

 496                hole = next(c for c in again["children"] if c["kind"] == "hole")
 497            inner = scene.node(low)
 498            hole["children"].append(inner)
 499            hole["edges"] += _through(inner, hole["inputs"], hole["outputs"])
 500            hole["badge"] = {"text": "⪯", "status": status}
 501            continue
 502        drawn = _find(nodes, scene.ids[id(high)]) if id(high) in scene.ids else None
 503        # a part of a composition is refined in place by its decomposition only;
 504        # another contract that refines it is drawn apart, and the composition stays
 505        decomposition = isinstance(low, System) or (
 506            isinstance(low, Contract) and bool(low.origin) and low.origin[0] == "composition")
 507        part = drawn is not None and (_parent(nodes, drawn["id"]) or {}).get("kind") == "group"
 508        loop = drawn is not None and any(scene.objs.get(n["id"]) == id(low)    # inside a box of low
 509                                         for n in _ancestors(nodes, drawn["id"]))
 510        if (drawn is not None and not drawn.get("children") and top.get(id(low)) is not drawn
 511                and (decomposition or not part) and not loop):
 512            # the contract is drawn already (inside another box, or as an item): its box takes what refines it
 513            inner = top.pop(id(low), None)
 514            if inner is not None:
 515                nodes.remove(inner)
 516            else:
 517                inner = scene.node(low)
 518            wrapped = scene.wrap(inner, high, status)
 519            keep = drawn["id"]
 520            drawn.clear()
 521            drawn.update(wrapped)
 522            drawn["id"] = keep
 523            for k, v in list(scene.ids.items()):
 524                if v == wrapped["id"]:
 525                    scene.ids[k] = keep
 526            continue
 527        if id(low) in refined:              # it refines another contract too: drawn again, apart
 528            after(top[id(low)], scene.wrap(scene.node(low), high, status))
 529            continue
 530        inner = top.pop(id(low), None) or next(   # in a chain, the box drawn already
 531            (n for n in nodes if n["id"] == scene.ids.get(id(low))), None) or scene.node(low)
 532        if _holds(inner, id(high), scene.objs):  # a loop (each refines the other): drawn apart
 533            top.setdefault(id(low), inner)
 534            after(inner, scene.wrap(scene.node(low), high, status))
 535            continue
 536        wrapped = scene.wrap(inner, high, status)
 537        replace(inner, wrapped)
 538        top[id(low)] = wrapped
 539        refined.add(id(low))
 540
 541    def box_of(obj) -> dict:              # an item drawn on its own first, then inside others
 542        if id(obj) in own:
 543            return own[id(obj)]
 544        if id(obj) not in scene.ids:
 545            return add(obj)
 546        return _find(nodes, scene.ids[id(obj)])
 547
 548    for c in checks:
 549        obj, kind, r = (c[0], None, c[1]) if len(c) == 2 else c
 550        node = box_of(obj)
 551        for label, res, core in _checks(r, kind):
 552            mark = {"name": label, "status": _mark(res), "reason": getattr(res, "reason", "") or ""}
 553            if core is not None:
 554                where = _core(node, core, unrealizable=label == "R")
 555                if where is None:
 556                    raise ValueError(f"the terms of the core are not those of {node['title']}")
 557                mark["core"] = where
 558            node.setdefault("checks", []).append(mark)
 559
 560    wires = []
 561    for e in environments:
 562        env, target = e[0], e[1]
 563        status = _status(e[2]) if len(e) > 2 else None
 564        node = box_of(target)
 565        env_node = scene.environment(env, node, status)
 566        outer = next(n for n in nodes if n is node or _find([n], node["id"]) is not None)
 567        nodes.insert(nodes.index(outer), env_node)
 568        if outer is node:                   # wires between boxes at the top only
 569            wires += [{"from": [env_node["id"], "out", v], "to": [node["id"], "in", v], "var": v}
 570                      for v in node.get("inputs", [])]
 571        else:
 572            env_node["badge"]["text"] = f"E ⊨ A of {node['title']}"
 573    for obj in open:                    # its first box, and its other boxes but the parts of a composition
 574        first = box_of(obj)
 575        boxes = [first] + [n for n in _boxes(nodes) if scene.objs.get(n["id"]) == id(obj) and n is not first
 576                           and (_parent(nodes, n["id"]) or {}).get("kind") != "group"]
 577        holes = [n for n in _boxes(nodes) if n["kind"] == "hole" and scene.objs.get(n["id"]) == id(obj)]
 578        for n in holes or boxes:        # a quotient: its holes, not the box of the dividend
 579            n["open"] = True
 580    if context is not None:
 581        from .logic import _formula
 582        context = [context] if isinstance(context, (str, ltl.Node)) else list(context)
 583        context = [ltl.to_str(_formula(f)) for f in context]
 584    return {"title": title or "CHASE", "nodes": nodes, "edges": wires, "context": context or []}

 628def from_spec(path: str, contracts: Iterable = (), refines: Iterable = (),
 629              environments: Iterable = (), check: bool = False, verify: bool = False,
 630              realizable: bool = False, timeout: float = 60, title: Optional[str] = None,
 631              open: Iterable = ()) -> dict:
 632    """The diagram of a specification file (the logics language of CHASE, or
 633    JSON). ``contracts`` are the contracts to draw (default: all), each a
 634    name or names joined by + for their composition; ``refines`` lists
 635    (refined, abstract) names, drawn one inside the other; ``environments``
 636    lists (model file, contract): an SMV or JSON state machine, or a
 637    controller of slugs. ``check`` checks the refinements and the
 638    environments; ``verify`` and ``realizable`` check each contract of the
 639    file that is drawn, alone or as a part of a composition. ``open`` names
 640    the contracts drawn open."""
 641    from . import checks as _checks_
 642    from . import gr1
 643    from .cli import _contract, _model, _read
 644    spec = _read(path)
 645    made = {}
 646
 647    def get(expression):              # one box per expression
 648        if expression not in made:
 649            made[expression] = _contract(expression, spec)
 650        return made[expression]
 651    items = [get(e) for e in (list(contracts) or list(spec))]
 652    pairs = [(get(a), get(b), _checks_.refines(get(a), get(b), timeout=timeout).status if check else None)
 653             for a, b in refines]
 654    def written(c, out):              # the contracts of the file in c, in order
 655        if c.origin:
 656            for o in c.origin[1]:
 657                written(o, out)
 658        elif all(x is not c for x in out):
 659            out.append(c)
 660        return out
 661    atoms = []
 662    for c in items + [x for a, b, _ in pairs for x in (a, b)]:
 663        written(c, atoms)
 664    marks = []
 665    for c in atoms:                   # a composition is not GR(1): its parts are checked
 666        if verify:
 667            marks.append((c, _checks_.verify(c, core=True, timeout=timeout)))
 668        if realizable:
 669            marks.append((c, "realizable", gr1.is_realizable(c, core=True, timeout=timeout)))
 670    envs = []
 671    for model, name in environments:
 672        c = get(name)
 673        m = _model(model, c)
 674        envs.append((m, c, _checks_.is_environment(m, c, timeout=timeout).status if check else None))
 675    return diagram(*items, refines=pairs, checks=marks, environments=envs,
 676                   open=[get(e) for e in open], title=title or os.path.basename(path))

 685def serve(page, path: str, port: int = 8765, host: str = "127.0.0.1") -> int:
 686    """Serves page() on http://host:port, and draws it again whenever the
 687    file at path changes; the page in the browser reloads by itself. Stops
 688    with Ctrl-C."""
 689    import http.server
 690    import threading
 691    import time
 692
 693    state = {"mtime": None, "version": 0, "page": ""}
 694    lock = threading.Lock()
 695
 696    def refresh():
 697        mtime = os.path.getmtime(path) if os.path.exists(path) else None
 698        if mtime == state["mtime"]:
 699            return
 700        try:
 701            text = page()
 702        except Exception as e:          # an error in the file is shown in the page
 703            text = ("<!doctype html><meta charset=utf-8><pre style='color:#b03a2e;padding:16px'>"
 704                    + _html.escape(f"{type(e).__name__}: {e}") + "</pre>" + _RELOAD)
 705        with lock:
 706            state.update(mtime=mtime, page=text, version=state["version"] + 1)
 707
 708    class Handler(http.server.BaseHTTPRequestHandler):
 709        def do_GET(self):
 710            refresh()
 711            with lock:
 712                body = (str(state["version"]) if self.path == "/version" else state["page"]).encode()
 713            self.send_response(200)
 714            self.send_header("Content-Type", "text/plain" if self.path == "/version"
 715                             else "text/html; charset=utf-8")
 716            self.send_header("Cache-Control", "no-store")
 717            self.end_headers()
 718            self.wfile.write(body)
 719
 720        def log_message(self, *args):
 721            pass
 722
 723    refresh()
 724    server = http.server.ThreadingHTTPServer((host, port), Handler)
 725    print(f"Drawing {path} on http://{host}:{port} (Ctrl-C to stop)")
 726    try:
 727        server.serve_forever()
 728    except KeyboardInterrupt:
 729        pass
 730    finally:
 731        server.server_close()
 732    return 0
 733

 736_RELOAD = """<script>
 737(function () {
 738  if (!location.protocol.startsWith("http")) return;
 739  let seen = null;
 740  setInterval(async () => {
 741    try { const v = await (await fetch("/version", { cache: "no-store" })).text();
 742          if (seen === null) seen = v; else if (v !== seen) location.reload(); } catch (e) {}
 743  }, 1000);
 744})();

 876  while (rest.length > WIDTH) {
 877    let cut = -1;
 878    for (const op of [" & ", " | ", " -> ", " <-> ", " "]) {
 879      const i = rest.lastIndexOf(op, WIDTH); if (i > WIDTH / 3) { cut = i; break; } }

 882  }
 883  out.push(rest); return out;

1026// The column of each box inside n, by the flow of its wires: the longest
1027// path from the inputs of n, a wire back to an earlier box (a loop) left out.
1028// Viewpoints stacked in one box share a column.
1029function columns(n) {
1030  const col = {}, ids = n.children.map(c => c.id);
1031  if (n.layout === "stack") { ids.forEach(id => { col[id] = 0; }); return col; }
1032  const next = {};
1033  ids.forEach(id => { next[id] = []; });
1034  (n.edges || []).forEach(w => { if (w.from && w.from[0] && w.to[0] && next[w.from[0]] && w.from[0] !== w.to[0])
1035    next[w.from[0]].push(w.to[0]); });
1036  const state = {}, order = [];
1037  const visit = id => { state[id] = 1;            // depth first, in the order of the list
1038    next[id] = next[id].filter(t => state[t] !== 1);  // a wire back to a box on the path: a loop
1039    next[id].forEach(t => { if (!state[t]) visit(t); });
1040    state[id] = 2; order.unshift(id); };
1041  ids.forEach(id => { if (!state[id]) visit(id); });
1042  order.forEach(id => { col[id] = col[id] || 0; next[id].forEach(t => { col[t] = Math.max(col[t] || 0, col[id] + 1); }); });
1043  return col;
1044}
1045// The groups of boxes inside n that are wired to each other (the ports of
1046// n left out): null when there is one group, or when n stacks viewpoints.
1047function bands(n) {
1048  if (n.layout === "stack" || n.children.length < 2) return null;
1049  const up = {};
1050  const root = id => up[id] === id ? id : (up[id] = root(up[id]));
1051  n.children.forEach(c => { up[c.id] = c.id; });
1052  (n.edges || []).forEach(w => { if (w.from && w.from[0] && w.to[0] && w.from[0] in up && w.to[0] in up)
1053    up[root(w.from[0])] = root(w.to[0]); });
1054  const rows = new Map();
1055  n.children.forEach(c => { const r = root(c.id); if (!rows.has(r)) rows.set(r, []); rows.get(r).push(c); });
1056  return rows.size > 1 ? [...rows.values()] : null;
