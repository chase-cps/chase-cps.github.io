24def register(name: str, adapter: Adapter, first: bool = False) -> None:
25    """Add an adapter (e.g. for another model checker)."""
26    global ADAPTERS
27    if first:
28        ADAPTERS = {name: adapter, **{k: v for k, v in ADAPTERS.items() if k != name}}
29    else:
30        ADAPTERS[name] = adapter
31

33def adapters_for(query) -> list[str]:
34    return [n for n, a in ADAPTERS.items() if isinstance(query, a.answers)]

37def check(query, backend: str = "auto", **options) -> Result:
38    """Answer a query with the given back end, or the first available one.
39
40    With backend "auto", the environment variable CHASE_BACKEND, when set,
41    names the back end to use for the queries it answers.
42    """
43    preferred = os.environ.get("CHASE_BACKEND")
44    if backend == "auto" and preferred and preferred in ADAPTERS \
45            and isinstance(query, ADAPTERS[preferred].answers):
46        backend = preferred
47    if backend != "auto":
48        if backend not in ADAPTERS:
49            raise ValueError(f"unknown back end {backend!r}; known: {sorted(ADAPTERS)}")
50        adapter = ADAPTERS[backend]
51        if not isinstance(query, adapter.answers):
52            return Result(UNKNOWN, backend,
53                          reason=f"{backend} does not answer {type(query).__name__}")
54        return _configured(adapter, options).run(query)
55    candidates = adapters_for(query)
56    if not candidates:
57        return Result(UNKNOWN, reason=f"no back end answers {type(query).__name__} yet")
58    for name in candidates:
59        if ADAPTERS[name].available():
60            return _configured(ADAPTERS[name], options).run(query)
61    return Result(UNKNOWN, reason="no installed back end answers "
62                  f"{type(query).__name__}; tried {candidates}. Run `chase-cps doctor` "
63                  "to see how to install them")
64
65
66def _configured(adapter: Adapter, options: dict) -> Adapter:
67    if not options:
68        return adapter
69    import copy
70    a = copy.copy(adapter)
71    for k, v in options.items():
72        if not hasattr(a, k):
73            raise TypeError(f"{adapter.name} has no option {k!r}")
74        setattr(a, k, v)
75    return a

 30    def __init__(self, tool: str = "auto", engine: str = "auto",
 31                 timeout: float = 300.0, bmc_bound: int = 20):
 32        self.tool = tool
 33        self.engine = engine
 34        self.timeout = timeout

 39    def binary(self) -> tuple[str, str] | None:
 40        """(path, flavor) of the model checker, flavor in {"nuxmv", "nusmv"}."""
 41        candidates = []
 42        if os.environ.get("CHASE_NUXMV"):
 43            candidates.append((os.environ["CHASE_NUXMV"], "nuxmv"))
 44        if os.environ.get("CHASE_NUSMV"):
 45            candidates.append((os.environ["CHASE_NUSMV"], "nusmv"))
 46        candidates += [("nuXmv", "nuxmv"), ("NuSMV", "nusmv")]
 47        for path, flavor in candidates:
 48            if self.tool not in ("auto", flavor):
 49                continue
 50            found = find(path)
 51            if found:
 52                return found, flavor
 53        return None

 55    def available(self) -> bool:
 56        return self.binary() is not None

 58    # emit / run ----------------------------------------------------------------
 59
 60    def emit(self, query) -> str:
 61        """The SMV model whose LTLSPEC is valid iff the query holds."""
 62        if isinstance(query, Implies):
 63            spec = ltl.Binary("->", query.phi, query.psi)
 64            variables = query.variables
 65        elif isinstance(query, Satisfiable):
 66            spec = ltl.Unary("!", query.phi)   # phi is satisfiable iff !phi is not valid
 67            variables = query.variables
 68        else:
 69            raise TypeError(f"{self.name} does not answer {type(query).__name__}")
 70        used = ltl.variables(spec)
 71        lines = ["MODULE main", "VAR"]
 72        for v, t in variables.items():
 73            if v in used:
 74                lines.append(f"    {v} : {'boolean' if t == 'bool' else f'{t[0]}..{t[1]}'};")
 75        if len(lines) == 2:   # no variable: NuSMV needs at least one
 76            lines.append("    chase_dummy : boolean;")
 77        lines.append(f"LTLSPEC {ltl.to_smv(spec)};")
 78        return "\n".join(lines) + "\n"
 79
 80    def run(self, query) -> Result:
 81        found = self.binary()
 82        if found is None:
 83            return Result(UNKNOWN, self.name, reason="neither nuXmv nor NuSMV is installed")
 84        path, flavor = found
 85        if self.engine != "auto":
 86            return self._run(query, path, flavor, self.engine, self.timeout)
 87        if flavor == "nuxmv":
 88            return self._run(query, path, flavor, "ic3", self.timeout)
 89        # NuSMV has no IC3: BMC finds counterexamples quickly, BDD proves validity.
 90        first = self._run(query, path, flavor, "bmc", min(self.timeout, 30.0))
 91        if first.status in (INVALID, SAT):
 92            return first
 93        result = self._run(query, path, flavor, "bdd", max(self.timeout - first.seconds, 1.0))
 94        result.seconds += first.seconds
 95        return result

 97    def _run(self, query, path: str, flavor: str, engine: str, timeout: float) -> Result:
 98        model = self.emit(query)
 99        start = time.monotonic()
100        with tempfile.TemporaryDirectory(prefix="chase_cps_") as tmp:
101            smv = os.path.join(tmp, "query.smv")
102            with open(smv, "w") as f:
103                f.write(model)
104            if flavor == "nuxmv":
105                commands = ["read_model -i query.smv", "flatten_hierarchy",
106                            "encode_variables", "build_boolean_model"]
107                if engine == "ic3":
108                    commands.append("check_ltlspec_ic3")
109                elif engine == "bmc":
110                    commands += ["bmc_setup", f"check_ltlspec_bmc -k {self.bmc_bound}"]
111                else:
112                    commands += ["build_model", "check_ltlspec"]
113                commands.append("quit")
114                with open(os.path.join(tmp, "commands"), "w") as f:
115                    f.write("\n".join(commands) + "\n")
116                args = [path, "-source", "commands"]
117            elif engine == "bmc":
118                args = [path, "-bmc", "-bmc_length", str(self.bmc_bound), "query.smv"]
119            else:
120                args = [path, "-dynamic", "-coi", "query.smv"]
121            try:
122                proc = subprocess.run(args, cwd=tmp, capture_output=True, text=True,
123                                      timeout=timeout)
124            except subprocess.TimeoutExpired:
125                return Result(UNKNOWN, f"{flavor}/{engine}",
126                              reason=f"timeout after {timeout:g} s",
127                              seconds=time.monotonic() - start)
128        seconds = time.monotonic() - start
129        backend = f"{flavor}/{engine}"
130        verdict = re.search(r"specification .* is (true|false)", proc.stdout, re.S)
131        if not verdict:
132            reason = (proc.stderr or proc.stdout).strip().splitlines()
133            reason = next((l for l in reason if "rror" in l), reason[-1] if reason else "")
134            if engine == "bmc" and "no counterexample found" in proc.stdout:
135                reason = f"no counterexample up to bound {self.bmc_bound}"
136            return Result(UNKNOWN, backend, reason=reason or "no verdict in the output",
137                          seconds=seconds, details={"model": model})
138        holds = verdict.group(1) == "true"
139        trace = None if holds else parse_trace(proc.stdout, query.variables)
140        if isinstance(query, Implies):
141            status = VALID if holds else INVALID
142        else:   # Satisfiable: !phi valid means phi unsatisfiable
143            status = UNSAT if holds else SAT
144        return Result(status, backend, trace=trace, seconds=seconds)

 18class SlugsAdapter(Adapter):
 19    """``controller``: also extract the controller when realizable;
 20    ``counter_strategy``: also extract the counter-strategy of the environment

 23    name = "slugs"
 24    answers = (Realizable,)
 25
 26    def __init__(self, controller: bool = False, counter_strategy: bool = True,
 27                 timeout: float = 300.0):
 28        self.controller = controller
 29        self.counter_strategy = counter_strategy
 30        self.timeout = timeout
 31
 32    def binary(self) -> str | None:
 33        return find(os.environ.get("CHASE_SLUGS", "slugs"))
 34
 35    def compiler(self) -> str | None:
 36        """The structured-slugs compiler shipped with slugs."""
 37        env = os.environ.get("SLUGSCOMPILER")
 38        if env and os.path.isfile(env):
 39            return env
 40        binary = self.binary()
 41        if binary:
 42            root = os.path.dirname(os.path.dirname(os.path.realpath(binary)))
 43            path = os.path.join(root, "tools", "StructuredSlugsParser", "compiler.py")
 44            if os.path.isfile(path):
 45                return path
 46        return None
 47
 48    def available(self) -> bool:
 49        return self.binary() is not None and self.compiler() is not None
 50
 51    def emit(self, query: Realizable) -> str:
 52        """The structured-slugs specification, written by the CHASE back end."""
 53        from .._native import pychase_backends
 54        pc = query.contract.to_pychase()
 55        with tempfile.TemporaryDirectory(prefix="chase_cps_") as tmp:
 56            path = os.path.join(tmp, "spec.structuredSlugs")
 57            pychase_backends.SlugsPrinter().print(pc, path)
 58            with open(path) as f:
 59                return f.read()
 60
 61    def _slugs(self, args: list[str], cwd: str) -> subprocess.CompletedProcess:
 62        return subprocess.run([self.binary()] + args, cwd=cwd, capture_output=True,
 63                              text=True, timeout=self.timeout)
 64
 65    def run(self, query: Realizable) -> Result:
 66        if not self.available():
 67            return Result(UNKNOWN, self.name, reason="slugs or its compiler.py is not installed; run "
 68                          "`chase-cps doctor` to see how to install it")
 69        start = time.monotonic()
 70        try:
 71            structured = self.emit(query)
 72        except RuntimeError as e:   # the contract is not in a GR(1) form
 73            return Result(UNKNOWN, self.name, reason=f"cannot encode for slugs: {e}")
 74        with tempfile.TemporaryDirectory(prefix="chase_cps_") as tmp:
 75            src = os.path.join(tmp, "spec.structuredSlugs")
 76            with open(src, "w") as f:
 77                f.write(structured)
 78            compiled = subprocess.run([sys.executable, self.compiler(), src],
 79                                      capture_output=True, text=True)
 80            if compiled.returncode != 0 or not compiled.stdout.strip():
 81                return Result(UNKNOWN, self.name, reason="structured slugs compilation failed: "
 82                              + compiled.stderr.strip()[-300:])
 83            slugsin = os.path.join(tmp, "spec.slugsin")
 84            with open(slugsin, "w") as f:
 85                f.write(compiled.stdout)
 86            try:
 87                proc = self._slugs(["spec.slugsin"], tmp)   # realizability only
 88                out = proc.stdout + proc.stderr
 89                if "is realizable" in out:
 90                    status = REALIZABLE
 91                elif "is unrealizable" in out:
 92                    status = UNREALIZABLE
 93                else:
 94                    return Result(UNKNOWN, self.name, reason=out.strip()[-300:],
 95                                  seconds=time.monotonic() - start)
 96                result = Result(status, self.name, details={"structuredSlugs": structured})
 97                if self.controller and status == REALIZABLE:
 98                    s = self._slugs(["--explicitStrategy", "--jsonOutput", "spec.slugsin"], tmp)
 99                    try:
100                        result.controller = json.loads(s.stdout)
101                    except json.JSONDecodeError:
102                        result.controller = s.stdout
103                elif self.counter_strategy and status == UNREALIZABLE:
104                    s = self._slugs(["--counterStrategy", "spec.slugsin"], tmp)
105                    result.counter_strategy = s.stdout
106            except subprocess.TimeoutExpired:
107                return Result(UNKNOWN, self.name, reason=f"timeout after {self.timeout:g} s",
108                              seconds=time.monotonic() - start)
109        result.seconds = time.monotonic() - start
110        return result
